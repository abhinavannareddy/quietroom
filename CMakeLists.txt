cmake_minimum_required(VERSION 3.16)
project(quietroom VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()

option(QUIETROOM_BUILD_TESTS "Build unit tests" ON)
option(QUIETROOM_BUILD_BENCH "Build Google Benchmark targets" ON)
option(QUIETROOM_DOUBLE "Use double instead of float for samples (F9)" OFF)
option(QUIETROOM_WERROR "Treat warnings as errors" ON)
option(QUIETROOM_SANITIZE "Enable ASan + UBSan" OFF)

# Warnings and sanitizers apply to our own targets only, never to fetched dependencies.
function(quietroom_target_options target)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /permissive-)
    if(QUIETROOM_WERROR)
      target_compile_options(${target} PRIVATE /WX)
    endif()
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wshadow -Wconversion)
    if(QUIETROOM_WERROR)
      target_compile_options(${target} PRIVATE -Werror)
    endif()
    if(QUIETROOM_SANITIZE)
      target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
  endif()
endfunction()

add_library(quietroom src/version.cpp src/framing.cpp)
target_include_directories(quietroom PUBLIC include)
if(QUIETROOM_DOUBLE)
  target_compile_definitions(quietroom PUBLIC QUIETROOM_DOUBLE=1)
endif()
quietroom_target_options(quietroom)

include(FetchContent)
set(FETCHCONTENT_QUIET OFF)

if(QUIETROOM_BUILD_TESTS)
  enable_testing()
  set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
  set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
  FetchContent_Declare(googletest
    URL https://github.com/google/googletest/archive/refs/tags/v1.15.2.tar.gz)
  FetchContent_MakeAvailable(googletest)
  add_subdirectory(tests)
endif()

if(QUIETROOM_BUILD_BENCH)
  set(BENCHMARK_ENABLE_TESTING OFF CACHE BOOL "" FORCE)
  set(BENCHMARK_ENABLE_GTEST_TESTS OFF CACHE BOOL "" FORCE)
  set(BENCHMARK_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(benchmark
    URL https://github.com/google/benchmark/archive/refs/tags/v1.9.1.tar.gz)
  FetchContent_MakeAvailable(benchmark)
  add_subdirectory(bench)
endif()
